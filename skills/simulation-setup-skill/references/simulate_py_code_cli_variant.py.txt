def _fetch_shadow_path() -> str:
    """Absolute path to the fetch-shadow preload, resolved from the INSTALLED plurai
    package — NEVER the customer harness (`_DIR`).

    `NODE_OPTIONS=--import` needs an absolute specifier (it resolves in the grandchild's
    cwd-independent context). Pointing it at the installed package keeps the preload TRIO
    co-located — `fetch_shadow.mjs` relatively-requires `./code_hook.cjs`, which registers
    `./code_hook_loader.mjs` — so the require-chain stays intact, AND keeps the path's
    dirname out of `plurai_sim/` (no copy: a neutral-named copy would break the relative
    require unless the whole trio moved). Fail closed if the package layout is missing.
    """
    path = Path(plurai.__file__).resolve().parent / '_mocking' / 'node' / 'fetch_shadow.mjs'
    if not path.is_file():
        raise FileNotFoundError(
            f'fetch_shadow.mjs not found in installed plurai package at {path} — '
            'cannot inject the cli_http preload (is plurai installed with _mocking/node/?)')
    return str(path)

@contextlib.contextmanager
def _withheld_env(names: tuple[str, ...]):
    """Withhold `names` from the environment the nested CLI starts with, and only then.

    claude_agent_sdk starts the CLI with this process's environment and lays `options.env` OVER it,
    so a key left out of `options.env` still reaches the agent's shell: the only way to withhold one is
    to take it out of this process's environment while the CLI starts. Put back after, for this
    process's own calls. Race-free: this process runs one scenario and handles its envelopes one at
    a time, so nothing else reads the environment while the CLI starts. A child factory that ran
    scenarios concurrently in one process would reopen that race.
    """
    held = {name: os.environ.pop(name) for name in names if name in os.environ}
    try:
        yield
    finally:
        os.environ.update(held)


def _isolated_agent_cwd() -> str:
    """Give the nested session a working directory that holds ONLY the production
    skill surface the real user drives — never the harness files (simulate.py,
    code_surface.yaml, simulation_env/mocks.py, plurai.config.yaml, sitecustomize.py).

    OPACITY: a Bash-bearing nested agent whose cwd is (or neighbours) the harness
    dir can `cat` the mock source / channel config and discover it is in a
    simulation — then it reverse-engineers the mock instead of exercising the real
    skill. We copy the sibling `skills/` tree into a fresh tempdir and run the
    agent from `<tmp>/work`, so the documented
    `node ../skills/<name>/scripts/cli.mjs` invocation still resolves while the
    harness dir (`_DIR`) is not reachable by any relative path.

    The env-var path pivot is CLOSED by `_relocate_surface` / `_fetch_shadow_path`
    / `_nested_env`: `$PLURAI_CODE_SURFACE` points at a clean per-session shim dir (a
    SEPARATE tempdir root from this cwd) and `NODE_OPTIONS=--import` points at the
    INSTALLED plurai package — neither into `_DIR` — so
    `cat "$(dirname "$PLURAI_CODE_SURFACE")"/simulation_env/mocks.py` does not reach
    the customer's mock answers.

    RESIDUAL — NOT closed here (Option-D tier, an out-of-trust-domain runner): a
    Bash-bearing agent can still env-dump and `cat` the shim's surface YAML / the
    installed plurai runtime (`fetch_shadow.mjs` / `code_hook.cjs`) — learning "this is
    a plurai sim" but NOT the customer's specific mock answers — dump a grandchild's
    `os.environ` / `/proc/<pid>/environ`, or hit the localhost mock URL. Blocking the
    `Read` tool is cosmetic while Bash is present.
    """
    import atexit
    import shutil
    import tempfile
    root = Path(tempfile.mkdtemp(prefix='plurai_agent_'))
    atexit.register(shutil.rmtree, root, ignore_errors=True)  # bounded: one dir per session
    skills_src = _DIR.parent / 'skills'                  # the vendored production skill(s)
    if skills_src.is_dir():
        shutil.copytree(skills_src, root / 'skills')     # ../skills/<name>/... stays resolvable
    workdir = root / 'work'
    workdir.mkdir()
    return str(workdir)

def _relocate_surface() -> str:
    """Copy `code_surface.yaml` into a clean per-session shim dir and return the copy.

    The shim is its OWN `tempfile.mkdtemp` root, kept SEPARATE from
    `_isolated_agent_cwd`'s root, so `dirname $PLURAI_CODE_SURFACE` resolves to the shim
    (which holds ONLY the surface), never the customer harness `_DIR`. Called from
    `seed_state` AFTER the module-level `emit_code_surface` block, so `_CODE_SURFACE`
    exists before the copy. The shim is atexit-cleaned like the cwd. (The Node preload is
    NOT copied — it stays in the installed package; see `_fetch_shadow_path`.)

    The surface copy is recorded once written (`plurai.watch_interception`): an agent that rewrites
    it switches the shadow off, and the run then reports it. The preload stays unrecorded: it is the
    installed package every bench shares, which an upgrade between runs changes for every row.

    HONEST SCOPE: this closes the CUSTOMER-mocks.py pivot only. The shim still holds the
    surface YAML; env-dumping and `cat`-ing it (generic-plurai detection) is an Option-D
    residual, not closed here.
    """
    import atexit
    import shutil
    import tempfile
    shim = tempfile.mkdtemp(prefix='plurai_shim_')          # SEPARATE root from the isolated cwd
    atexit.register(shutil.rmtree, shim, ignore_errors=True)  # bounded: one dir per session
    shim_surface = str(Path(shim) / 'surface.yaml')         # neutral name (no `code_surface` tell in basename)
    shutil.copyfile(_CODE_SURFACE, shim_surface)            # surface exists (emitted at import) — re-readable copy
    plurai.watch_interception(shim_surface)                 # this scenario's surface copy, as written
    return shim_surface


def _nested_env(sid: str, mock_url: str, shim_surface: str) -> dict[str, str]:
    """Env for the nested session's Bash subprocesses (the `node cli.mjs ...` runs): ONLY what
    the fetch-shadow preload needs.

    claude_agent_sdk lays this over the environment the CLI starts with, so the rest of this
    process's environment reaches the agent's shell as it is, less `_WITHHELD_ENV`
    (`_withheld_env` around `connect()`). The preload is
    loaded into every `node` grandchild via NODE_OPTIONS=--import BEFORE the CLI's first
    fetch — pointed at the INSTALLED plurai package
    (`_fetch_shadow_path`), so the require-trio stays intact and no `--import` path roots
    in `plurai_sim/`. `PLURAI_CODE_SURFACE` points at the shim copy
    (`_relocate_surface`), so `dirname $PLURAI_CODE_SURFACE` is the shim, never the
    harness. `sid` and `mock_url` come from the LIVE session (resolved in seed_state via
    plurai accessors; `mock_url` is `plurai.mock_service_url()`, the mock url this harness dials,
    and the bearer rides beside it), NOT from os.environ.

    HONEST SCOPE: relocating the paths closes the CUSTOMER-mocks.py dirname pivot. It does
    NOT make the run undetectable — `env` in the persistent shell still shows `PLURAI_*`, a
    grandchild `os.environ` dump still returns these values, and the installed-plurai
    runtime + surface YAML are still `cat`-able (generic-plurai detection). Those are
    Option-D residuals (out-of-trust-domain runner).
    """
    env = {}
    env['PLURAI_SESSION_ID'] = sid                       # live session (read in seed_state)
    env['PLURAI_MOCK_URL'] = mock_url                    # the mock base url the agent's processes dial
    env['PLURAI_CODE_SURFACE'] = shim_surface            # shim copy — dirname is the shim, NOT plurai_sim/
    # The mock's bearer, the one this harness's own mock client sends: the agent's processes dial the
    # mock url directly, and the interception hooks read the bearer from their environment.
    if os.environ.get('PLURAI_TOKEN'):
        env['PLURAI_TOKEN'] = os.environ['PLURAI_TOKEN']
    # Inject the preload into every node grandchild. Preserve any inherited NODE_OPTIONS.
    _import = f'--import={_fetch_shadow_path()}'          # installed-package preload (absolute, outside _DIR)
    env['NODE_OPTIONS'] = ' '.join(
        ([os.environ['NODE_OPTIONS']] if os.environ.get('NODE_OPTIONS') else []) + [_import])
    # Head PATH with THIS bench interpreter's bindir so the grandchild's `node` (and any
    # `python`) resolves to the venv that carries the plurai runtime — parity with the SDK
    # variant. `node` is usually system-wide, but a system python lacking plurai would
    # otherwise break a mixed python+node skill's interception.
    env['PATH'] = os.pathsep.join(
        [os.path.dirname(sys.executable)] + ([os.environ['PATH']] if os.environ.get('PATH') else []))
    # The agent's shell is not network-sandboxed: what keeps the CLI's calls on the mock is the
    # preload's `unmapped_egress: fail_closed` (any fetch to a host/URL not in cli_http aborts before
    # leaving) and the withheld creds (a call that slips past it cannot authenticate).
    return env


def _skill_source_dir_name() -> str | None:
    """The skill-dir name recorded as the command under test — `agent.skill_source` in
    `plurai.config.yaml` (S1), e.g. `skills/<name>/SKILL.md` -> `<name>`. Used ONLY to
    disambiguate when more than one skill dir is vendored; returns `None` if the config is
    unreadable or `skill_source` is not a `skills/<name>/...` path (e.g. a `commands/*.md`).
    """
    try:
        import yaml  # noqa: PLC0415 — available in the bench env (plurai depends on it)
        data = yaml.safe_load(_CONFIG.read_text(encoding='utf-8'))
    except Exception:
        return None
    src = ((data or {}).get('agent') or {}).get('skill_source', '') if isinstance(data, dict) else ''
    parts = Path(str(src)).parts
    if 'skills' in parts:
        i = parts.index('skills')
        if i + 1 < len(parts):
            return parts[i + 1]
    return None


def _agent_skill_dir() -> str:
    """Resolve the `{skill_dir}` token the rendered `skill_prompt.md` carries (S2's `code`
    path-fact rule) to the runtime-reachable RELATIVE path the isolated cwd exposes the
    vendored skill at: `../skills/<name>`.

    `<name>` is the skill dir UNDER TEST, identified WITHOUT assuming what the vendored
    command is named or where it lives — the command need not be `scripts/cli.*`; S2 rule 5
    substitutes only the DIRECTORY token, so the command markdown supplies the rest of the
    path (`scripts/cli.mjs`, `bin/run.js`, `dispatch.py`, ...) after `{skill_dir}`.
    Resolution order:
      1. exactly ONE vendored dir under `../skills` -> that one (the common bench shape);
      2. else the dir named by `agent.skill_source` in `plurai.config.yaml` (the command
         recorded under test at S1);
      3. else WARN and fall back to the first dir — the smoke's >=1-tool_event check is the
         backstop for a wrong pick.

    `_isolated_agent_cwd()` copies the sibling `skills/` tree next to the agent's
    `<tmp>/work` cwd, so `../skills/<name>/...` resolves at runtime.

    OPACITY: this ALWAYS returns a RELATIVE `../skills/<name>` — NEVER an absolute
    `_DIR`-rooted path. `../skills/<name>` is the sanctioned reachable surface (asserted by
    tests/test_harness_opacity.py); the harness dir (`_DIR`) must stay unreachable.
    """
    skills_src = _DIR.parent / 'skills'
    subdirs = (sorted(p.name for p in skills_src.iterdir() if p.is_dir())
               if skills_src.is_dir() else [])
    if not subdirs:
        print('[plurai] WARNING: no vendored skill dir under ../skills; {skill_dir} left '
              'as ../skills (the command path will be wrong) — check the vendored layout.',
              file=sys.stderr)
        return '../skills'
    if len(subdirs) == 1:                        # common bench: exactly one vendored skill
        return f'../skills/{subdirs[0]}'
    chosen = _skill_source_dir_name()            # disambiguate by the recorded command under test
    if chosen in subdirs:
        return f'../skills/{chosen}'
    print(f'[plurai] WARNING: {len(subdirs)} vendored skill dirs {subdirs} and '
          f'agent.skill_source ({chosen!r}) did not identify one; {{skill_dir}} resolved to '
          f'../skills/{subdirs[0]} (first) — verify the rendered command path in the smoke.',
          file=sys.stderr)
    return f'../skills/{subdirs[0]}'


_RESUME_ROLE = {'human': 'user', 'ai': 'assistant', 'tool': 'tool', 'system': 'system'}

#: What ``seed_state`` returns after a resumed scenario. The SDK passes this value through
#: VERBATIM into the run record, where ``None`` reads as a full-fidelity resume — so a lane
#: that restores the transcript and nothing else has to say so, or the run is scored as though
#: the agent resumed from state it never saw. The transcript IS this lane's ceiling: the nested
#: session holds no store outside its own conversation, so every ``agent_state`` key the seed
#: carries is skipped and named under ``resume_state_dropped``.
_MESSAGES_ONLY_FIDELITY = {'fidelity': 'reduced', 'restored': ['messages'], 'not_restored': ['agent_state']}

def _resume_transcript(resume_seed: dict) -> str:
    """Render a resumed scenario's pinned transcript (minimal message dicts) as a
    system-prompt block, injected AT CONSTRUCTION — `seed_state` is where the nested
    session is built, and the context it opens on is the only place history can go.
    That is messages fidelity, and also the ceiling: the agent has no state outside
    its own conversation, so ``agent_state`` is not read here — ``seed_state`` names
    its keys in the fidelity report instead. A ``resume_point`` is refused: it
    names a graph node to start turn 1 at, and this lane has no nodes, so there is
    nothing to honour and nothing to reduce — starting from the top instead would
    redo the very steps the trace already took. (IDENTICAL across all three
    variants.)
    """
    point = resume_seed.get('resume_point')
    if point is not None:
        raise RuntimeError(
            f'resume_seed pins a start node — resume_point={point!r} — and a skill-lane '
            'harness has no graph node to start from — the nested session begins where its '
            'context begins. Re-convert the trace without --start-node, or target a graph '
            'harness. Refusing rather than starting from the top, which would redo the steps '
            'the trace already took and score that as the pinned resume.')
    messages = resume_seed.get('messages') or []
    if not messages:
        raise RuntimeError(
            'resume_seed.messages is empty — seeding nothing would run the scenario on '
            'a blank session and score a conversation the agent never had')
    lines = []
    for m in messages:
        role = _RESUME_ROLE.get(m.get('type'))
        if role is None:
            # Raise, never skip: a silently dropped exchange is a transcript that no
            # longer explains itself.
            raise RuntimeError(f'resume_seed message {m.get("id")!r} has unrenderable '
                               f'type {m.get("type")!r}')
        content = m.get('content')
        text = content if isinstance(content, str) else json.dumps(
            content, ensure_ascii=False, default=str)
        if role == 'tool':
            # tool_call_id is the in-transcript join between a call and its result.
            # Message ids are not rendered: the byte-for-byte resume seed this
            # session was handed (resume_seed.yaml, in the scenario folder) keeps
            # the join back to the source trace.
            lines.append(f'[tool result for {m.get("tool_call_id")}] {text}')
            continue
        lines.append(f'[{role}] {text}'.rstrip())
        for call in m.get('tool_calls') or []:
            args = json.dumps(call.get('args') or {}, ensure_ascii=False, default=str)
            lines.append(f'[{role} tool call {call.get("id")}] {call.get("name")}({args})')
    return (
        '\n\n## Resumed conversation (this session, before this point)\n\n'
        'This session continues a conversation already in progress. The transcript\n'
        'below already happened between you and the user — treat it as your own prior\n'
        'turns: do not greet the user afresh, do not redo the tool calls shown, and\n'
        'rely on it for context as the conversation continues.\n\n'
        + '\n'.join(lines) + '\n')


async def seed_state(
    initial_agent_context: dict,
    initial_world_state: dict,
    resume_seed: dict | None = None,   # always passed (None when fresh); folded in at construction below
) -> dict | None:
    """Create the persistent skill session — IDENTICAL to the SDK variant's seed_state
    except the nested env is the CLI one above. Read sid + mock_url from the live
    session here (never at import, never from os.environ — see the SDK variant notes).
    A non-None ``resume_seed`` is folded into the session's opening context at
    construction below (messages fidelity, this lane's ceiling); a seed's
    ``agent_state`` keys are skipped and named in the report
    (``resume_state_dropped``), and a ``resume_point`` pinning a graph node this
    lane cannot start at is refused in ``_resume_transcript``.

    On a resumed scenario this returns the harness's own fidelity report — the
    transcript went in, the agent state did not — which the SDK carries verbatim into
    the run record. A fresh scenario returns ``None``, the value that means full
    fidelity, so nothing changes for a run that never resumes.
    """
    global _client
    sid = plurai.active_session_id()
    if not sid:
        raise RuntimeError('no active plurai session id at seed_state — CLI code channel '
                           'cannot inject PLURAI_SESSION_ID into the cli.mjs env')
    mock_url = plurai.mock_service_url()                 # the mock url the agent's processes dial
    if not mock_url:
        raise RuntimeError('no plurai mock service url at seed_state — CLI code channel '
                           'cannot inject PLURAI_MOCK_URL (it comes from the live '
                           'IPC-configured session, not this harness env)')
    prompt = _PROMPT.replace('{room_state}',
        str(initial_agent_context.get('room_state') or 'No context yet.'))
    # Resolve the `{skill_dir}` token the rendered skill_prompt.md carries (S2's
    # code/CLI path-fact rule) to the runtime-reachable RELATIVE path `../skills/<name>`,
    # so the documented `node {skill_dir}/scripts/cli.mjs <sub> '{…}'` runs as written
    # from the isolated cwd. Parity with the `{room_state}` fill above; NEVER an absolute
    # `_DIR` path (opacity — see `_agent_skill_dir`).
    prompt = prompt.replace('{skill_dir}', _agent_skill_dir())
    fidelity = None
    if resume_seed is not None:
        # Resumed scenario: fold the pinned transcript into the context the session
        # opens on — construction happens right below. Kept AFTER the token fills
        # so customer transcript text is never token-rewritten.
        prompt += _resume_transcript(resume_seed)
        fidelity = _MESSAGES_ONLY_FIDELITY.copy()
        # The seed's state has no slot here: skipped, and named so the replay says so.
        if dropped := sorted(resume_seed.get('agent_state') or {}):
            fidelity['resume_state_dropped'] = dropped
    # Relocate the surface to a clean shim BEFORE building the env, so no injected
    # value's dirname/--import reaches the customer harness (`_DIR`).
    shim_surface = _relocate_surface()
    cwd = _isolated_agent_cwd()
    # What the agent reads becomes trail, and the trail leaves the machine: its reads are scoped,
    # and its shell starts from none of the operator's startup files.
    _scope = plurai.host_read_scope(run_dir=Path(cwd).parent, repo=_DIR.parent, harness=_DIR)  # work/ and its ../skills copy
    _client = ClaudeSDKClient(ClaudeAgentOptions(
        system_prompt=prompt,
        # `code` channel: an ALLOWLIST of Bash, so the skill can run the vendored cli.mjs,
        # and no other built-in (add 'Write' only if the skill's workflow authors a params
        # file); none of the operator's settings, plugins or MCP servers.
        tools=['Bash'],
        setting_sources=[],
        strict_mcp_config=True,
        permission_mode='bypassPermissions',   # creds withheld + fetch-shadow fails closed on unmapped egress
        model=os.environ.get('PLURAI_SKILL_MODEL', 'sonnet'),
        cwd=cwd,                               # ← isolated skill surface, NOT the harness dir (opacity)
        env={**_nested_env(sid, mock_url, shim_surface), **_scope.agent_shell_env(cli_env=os.environ)},  # ← CLI code-channel injection w/ relocated surface
        hooks=plurai.cli_hooks(_scope, trail=None),  # its reads gated; a turn the CLI ends in error typed by the SDK
    ))
    try:
        with _withheld_env(_WITHHELD_ENV):     # the CLI, and every shell it runs, starts without them
            await _client.connect()
    except Exception as exc:
        hint = ' (this harness drops ANTHROPIC_API_KEY; the nested CLI signs in with your Claude login — run `claude` and sign in)' if _had_api_key else ''
        raise RuntimeError(f'nested skill session failed to start{hint}: {exc}') from exc
    return fidelity


async def invoke_agent(payload: dict, config: dict | None = None) -> dict:
    # IDENTICAL to the `tools` variant — reproduce that body verbatim. (One
    # consolidated AIMessage per turn; the evidence lives in the mock's tool_events
    # recorded by the preload's /mock/execute calls, not in LangChain tool_calls;
    # raise on empty output. The CLI sub-mode changes only seed_state / _nested_env.)
    ...
